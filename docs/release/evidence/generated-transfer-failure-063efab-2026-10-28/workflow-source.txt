name: Android Transport Diagnostic

on:
  workflow_dispatch:
  pull_request:
    paths:
      - ".github/workflows/android-transport-probe.yml"
      - "scripts/diagnostics/adb-transfer-*.mjs"
      - "scripts/diagnostics/run-adb-transfer-probe.mjs"
      - "tests/adb-transfer-probe.test.mjs"

permissions:
  contents: read

jobs:
  generated-file-transfer:
    name: Generated-file transport (not release)
    runs-on: ubuntu-latest
    timeout-minutes: 12
    steps:
      - uses: actions/checkout@v6
      - uses: actions/setup-node@v6
        with:
          node-version: "22"
      - name: Test diagnostic boundaries without a device
        run: node --test tests/adb-transfer-probe.test.mjs
      - uses: android-actions/setup-android@v4
      - name: Enable disposable virtual-device acceleration
        run: |
          echo 'KERNEL=="kvm", GROUP="kvm", MODE="0666", OPTIONS+="static_node=kvm"' | sudo tee /etc/udev/rules.d/99-kvm4all.rules
          sudo udevadm control --reload-rules
          sudo udevadm trigger --name-match=kvm
      - name: Run fixed primary transfers without app accounts or provider requests
        uses: reactivecircus/android-emulator-runner@v2
        env:
          SGM_LOCAL_DISPOSABLE_EMULATOR: "true"
          ADB_TRACE: transport
        with:
          api-level: 35
          target: google_apis_playstore
          arch: x86_64
          profile: pixel_6
          emulator-options: -no-window -gpu software -no-snapshot -noaudio -no-boot-anim
          script: node scripts/diagnostics/run-adb-transfer-probe.mjs
      - name: Retain only sanitized command outcomes and trace counts
        if: ${{ always() }}
        uses: actions/upload-artifact@v7
        with:
          name: android-generated-transfer-diagnostic
          path: android/app/build/outputs/transport-probe/*.json
          if-no-files-found: error
