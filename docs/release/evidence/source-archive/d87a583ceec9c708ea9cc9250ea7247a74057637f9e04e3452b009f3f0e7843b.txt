rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn() &&
        request.auth.token.roleKeys != null &&
        request.auth.token.roleKeys.hasAny(['admin']);
    }

    function hasClubManagerRole() {
      return signedIn() &&
        request.auth.token.roleKeys != null &&
        request.auth.token.roleKeys.hasAny(['clubManager']);
    }

    function isSelf(userId) {
      return signedIn() && request.auth.uid == userId;
    }

    function hasAnyKey(fields) {
      return request.resource.data.keys().hasAny(fields);
    }

    function isClubManager(clubId) {
      return hasClubManagerRole() &&
        exists(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)) &&
        get(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)).data.userId == request.auth.uid &&
        get(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)).data.clubId == clubId &&
        get(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)).data.role == 'clubManager' &&
        get(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)).data.status == 'active' &&
        (!get(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)).data.keys().hasAny(['active']) ||
          get(/databases/$(database)/documents/memberships/$(request.auth.uid + '_' + clubId)).data.active != false);
    }

    function userProtectedFields() {
      return [
        'roles',
        'childInvitationHash',
        'verified',
        'driverVerified',
        'stripeAccountId',
        'stripeStatus',
        'createdAt',
        'updatedAt'
      ];
    }

    function childProtectedFields() {
      return [
        'guardianUserIds',
        'createdAt'
      ];
    }

    match /childAccountLinks/{accountUserId} {
      allow read, write: if false;
    }
    match /childAccountApprovalOwners/{accountUserId} {
      allow read, write: if false;
    }
    match /childAccountApprovalRequests/{codeHash} {
      allow read, write: if false;
    }
    match /childAccountInvitations/{invitationHash} {
      allow read, write: if false;
    }
    match /childAccountInvitationOwners/{ownerId} {
      allow read, write: if false;
    }
    match /childAccountInvitationAccounts/{accountUserId} {
      allow read, write: if false;
    }

    match /users/{userId} {
      allow read: if isSelf(userId) || isAdmin();
      allow create: if isSelf(userId) && !hasAnyKey(userProtectedFields());
      allow update: if !request.resource.data.diff(resource.data).affectedKeys().hasAny(['childInvitationHash']) && (isAdmin() || (
        isSelf(userId) &&
        !request.resource.data.diff(resource.data).affectedKeys().hasAny(userProtectedFields())
      ));
      allow delete: if isAdmin();
    }

    match /children/{childId} {
      allow read: if isAdmin() || (signedIn() && resource.data.guardianUserIds.hasAny([request.auth.uid]));
      allow create: if signedIn() && request.resource.data.guardianUserIds.hasAny([request.auth.uid]);
      allow update: if isAdmin() || (
        signedIn() &&
        resource.data.guardianUserIds.hasAny([request.auth.uid]) &&
        !request.resource.data.diff(resource.data).affectedKeys().hasAny(childProtectedFields())
      );
      allow delete: if isAdmin();
    }

    match /clubs/{clubId} {
      allow read: if true;
      allow create, update: if isAdmin() || isClubManager(clubId);
      allow delete: if isAdmin();
    }

    match /teams/{teamId} {
      allow read: if true;
      allow create: if isAdmin() || isClubManager(request.resource.data.clubId);
      allow update: if isAdmin() || (
        resource.data.clubId == request.resource.data.clubId &&
        isClubManager(resource.data.clubId)
      );
      allow delete: if isAdmin() || isClubManager(resource.data.clubId);
    }

    match /memberships/{membershipId} {
      allow read: if isAdmin() || isSelf(resource.data.userId) || isClubManager(resource.data.clubId);
      allow create, update, delete: if isAdmin();
    }

    match /emergencyContacts/{contactId} {
      allow read, write: if false;
    }

    match /emergencyContactOperations/{operationId} {
      allow read, write: if false;
    }

    match /guardianVerifications/{verificationId} {
      allow read: if isSelf(resource.data.parentUserId) || isClubManager(resource.data.clubId);
      allow write: if false;
    }

    match /guardianVerificationEvents/{eventId} {
      allow read: if isSelf(resource.data.parentUserId) || isClubManager(resource.data.clubId);
      allow write: if false;
    }

    match /trips/{tripId} {
      allow read: if signedIn();
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (signedIn() &&
        resource.data.driverUserId == request.auth.uid &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['title']));
    }

    match /bookings/{bookingId} {
      allow read: if isAdmin() ||
        resource.data.parentUserId == request.auth.uid ||
        resource.data.driverUserId == request.auth.uid;
      allow create, update, delete: if false;
    }

    match /rideSessions/{rideSessionId} {
      allow read: if isAdmin() ||
        (signedIn() && resource.data.driverUserId == request.auth.uid) ||
        (signedIn() && resource.data.participantUserIds.hasAny([request.auth.uid])) ||
        (signedIn() && resource.data.childUserIds.hasAny([request.auth.uid]));
      allow write: if false;
    }

    // Team history is served only through current-authority, admission-filtered callables.
    match /teamConversations/{document=**} {
      allow read, write: if false;
    }
    match /teamMessageOperations/{operationId} {
      allow read, write: if false;
    }

    match /messages/{messageId} {
      allow read: if signedIn() && resource.data.participantUserIds.hasAny([request.auth.uid]);
      allow create, update, delete: if false;
    }

    match /chatInboxes/{entryId} {
      allow read: if signedIn() && resource.data.userId == request.auth.uid;
      allow create, update, delete: if false;
    }

    match /notifications/{notificationId} {
      allow read: if signedIn() && resource.data.userId == request.auth.uid;
      allow create, update, delete: if false;
    }

    match /ratings/{ratingId} {
      allow read: if signedIn() && resource.data.authorUserId == request.auth.uid;
      allow create, update, delete: if false;
    }

    match /co2Ledger/{entryId} {
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow write: if false;
    }

    match /rewardLedger/{entryId} {
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow write: if false;
    }

    match /stripeAccounts/{accountId} {
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow write: if false;
    }

    match /reports/{reportId} {
      allow read: if isAdmin();
      allow create, update, delete: if false;
    }

    match /supportUrgentEmailOutbox/{reportId} {
      allow read: if isAdmin();
      allow write: if false;
    }

    match /appConfig/{configId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
