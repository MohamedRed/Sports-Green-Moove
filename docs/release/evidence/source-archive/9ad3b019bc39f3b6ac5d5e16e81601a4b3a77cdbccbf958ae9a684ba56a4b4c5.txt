name: Focused Android Safety UI

on:
  workflow_call:

jobs:
  focused-case:
    name: Isolated ${{ matrix.case }}
    runs-on: ubuntu-latest
    timeout-minutes: 25
    strategy:
      fail-fast: false
      matrix:
        case: [emergency-identity, ride-observation, native-startup, remote-ride-stop, session-restoration]
    permissions:
      contents: read
    steps:
      - uses: actions/checkout@v6
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: "21"
      - uses: actions/setup-node@v6
        with:
          node-version: "22"
          cache: npm
      - run: npm ci
      - uses: android-actions/setup-android@v4
      - name: Compile isolated app before starting the device
        run: ./android/gradlew -p android :app:assembleDebug :app:assembleDebugAndroidTest :qa-install-guard:assembleDebug -PSGM_USE_FIREBASE_EMULATOR=true --no-daemon --max-workers=2
      - name: Enable KVM group permissions
        run: |
          echo 'KERNEL=="kvm", GROUP="kvm", MODE="0666", OPTIONS+="static_node=kvm"' | sudo tee /etc/udev/rules.d/99-kvm4all.rules
          sudo udevadm control --reload-rules
          sudo udevadm trigger --name-match=kvm
      - name: Run one isolated safety flow on its own phone
        uses: reactivecircus/android-emulator-runner@v2
        env:
          SGM_LOCAL_DISPOSABLE_EMULATOR: "true"
          SGM_PAUSE_TEST_PHONE_STORE: "true"
          SGM_BLOCK_TEST_PHONE_INSTALLS: "true"
          SGM_TEST_PHONE_ADMIN_APPROVED: "true"
        with:
          api-level: 35
          target: google_apis_playstore
          arch: x86_64
          profile: pixel_6
          script: bash scripts/run-focused-android-ui.sh ${{ matrix.case }}
      - name: Retain focused outcomes and independent cleanup
        if: ${{ always() }}
        uses: actions/upload-artifact@v7
        with:
          name: android-focused-safety-ui-${{ matrix.case }}
          path: |
            android/app/build/outputs/focused-safety-ui/**
            android/app/build/outputs/local-ui-evidence/**
          if-no-files-found: ignore
  focused-safety-ui:
    name: Isolated safety flows
    if: ${{ always() }}
    needs: focused-case
    runs-on: ubuntu-latest
    timeout-minutes: 5
    permissions:
      contents: read
    steps:
      - name: Require every independent case and its cleanup to pass
        env:
          FOCUSED_RESULT: ${{ needs.focused-case.result }}
        run: test "$FOCUSED_RESULT" = success
