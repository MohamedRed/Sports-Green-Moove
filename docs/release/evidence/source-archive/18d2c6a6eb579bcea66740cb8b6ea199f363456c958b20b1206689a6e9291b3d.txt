import Foundation

@MainActor func checkEmergencyContacts() async {
    let (account, access) = emergencyAccess()
    let gateway = EmergencyContactProbe(value: isolatedEmergencyView())
    let state = EmergencyContactState(access: access, scope: .managed(.personal), gateway: gateway)
    let phone = EmergencyPhoneProbe()
    await state.refresh()
    check(state.loaded && state.view?.contact == nil && !state.canDial && state.canEdit, "Absent contact was presented as an available number")
    state.name = isolatedContact.name; state.phone = "0470000000"
    check(!state.canSave, "Local number accepted without a country decision")
    state.phone = isolatedContact.phone; state.reviewSave()
    check(state.pending?.contact == isolatedContact, "Review did not pin the displayed private contact")
    state.cancelReview(); await state.confirmSave()
    let cancelled = await gateway.operations
    check(cancelled.isEmpty, "Canceling review wrote a contact")
    state.reviewSave(); state.phone = "+32470000002"; await state.confirmSave()
    check(state.view?.contact == isolatedContact && state.view?.revision == 1 && !state.dirty, "Review intent was replaced by later input")
    state.name = "Unsaved change"
    check(!state.canDial, "Unreviewed editor changes left an unrelated saved destination actionable")
    state.discardEdits(); check(state.canDial && !state.dirty, "Explicitly discarding edits did not retain the saved contact")
    await state.dial(using: phone)
    check(phone.contacts == [isolatedContact], "Current contact was not handed to the controlled opener")
    phone.result = .unavailable; await state.dial(using: phone)
    check(state.error != nil && state.view?.contact == isolatedContact, "Unavailable phone lost the contact or claimed success")
    await state.refresh()
    let changed = PersonalEmergencyContact(name: "Autre contact fictif", phone: "+32470000003")
    await gateway.configure(isolatedEmergencyView(contact: changed, revision: 2))
    let before = phone.contacts.count; await state.dial(using: phone)
    check(phone.contacts.count == before && state.view?.contact == changed && state.notice != nil, "Changed contact opened without a second review tap")
    phone.result = .opened; await state.dial(using: phone)
    check(phone.contacts.last == changed, "Explicit second tap did not recheck and open the current destination")
    state.reviewSave(removing: true); state.cancelReview(); await state.confirmSave()
    check(state.view?.contact == changed, "Cancelled removal cleared the current contact")
    state.reviewSave(removing: true); await state.confirmSave()
    check(state.view?.contact == nil && state.view?.revision == 3 && !state.canDial, "Removal did not retain a null-contact revision")
    state.name = isolatedContact.name; state.phone = isolatedContact.phone; state.reviewSave()
    await gateway.loseNextWriteReply(); await state.confirmSave()
    let writes = await gateway.operations.count
    check(state.view == nil && state.error != nil && !state.canDial && !state.canSave, "Unknown save outcome remained actionable")
    await state.confirmSave(); let afterRetry = await gateway.operations.count
    check(afterRetry == writes, "Uncertain save was automatically resent")
    await state.refresh()
    check(state.view?.contact == isolatedContact && state.view?.revision == 4, "Explicit read did not recover the stored contact")
    state.name = "Unsaved"; state.suspend()
    check(state.name.isEmpty && state.phone.isEmpty && state.view == nil && !state.canDial, "Background retained private contact or unsaved fields")
    await state.resume(); check(state.loaded && state.view?.contact == isolatedContact, "Foreground did not require a fresh authoritative read")
    state.close(); let prior = await gateway.names(); await state.refresh(); await state.dial(using: phone)
    let afterClose = await gateway.names(); check(afterClose == prior, "Closed contact screen issued new work")
    check(!account.busy, "Completed emergency work retained account admission")
    await checkChildEmergencyContacts()
    await checkEmergencyContactBoundaries()
    print("EmergencyContactChecks passed: review/cancel/save/remove, lost reply without resend, changed contact second tap, unavailable phone and private lifecycle; controlled gateways only")
}

@MainActor private func checkChildEmergencyContacts() async {
    let (_, access) = emergencyAccess(uid: "child-login", roles: [.child])
    let child = EmergencyContactView(ownerUid: "child-login", scope: .childRide("ride"), contact: isolatedContact,
        revision: 1, childId: "child-profile-not-login", childInstanceId: String(repeating: "a", count: 64), childName: "Enfant fictif")
    let gateway = EmergencyContactProbe(value: child)
    let state = EmergencyContactState(access: access, scope: .childRide(nil), gateway: gateway)
    let phone = EmergencyPhoneProbe(); await state.refresh()
    check(state.loaded && state.canDial && !state.canEdit, "Linked child cannot read a contact without the location flow")
    state.name = "Unauthorised"; state.phone = "+32470000002"; state.reviewSave(); await state.confirmSave()
    let operations = await gateway.operations; check(operations.isEmpty && state.pending == nil, "Child account gained contact editing")
    var recreated = child; recreated.childInstanceId = String(repeating: "b", count: 64)
    await gateway.configure(recreated); await state.dial(using: phone)
    check(phone.contacts.isEmpty && state.notice != nil, "Recreated child inherited a prior phone handoff")
    await state.dial(using: phone); check(phone.contacts == [isolatedContact], "Fresh child identity was not reviewed on a second tap")
    await gateway.configure(nil); await state.refresh()
    check(state.loaded && state.view == nil && !state.canDial, "Absent current child ride was replaced with a contact")
    let forbidden = EmergencyContactState(access: access, scope: .managed(.personal), gateway: gateway)
    await forbidden.refresh(); check(!forbidden.canEdit && forbidden.view == nil, "Mixed child role gained personal editing")
    let mixed = AccountWorkScope()
    mixed.replace(session: AuthSession(uid: "child-login", email: nil, displayName: nil, roles: [.child, .parent]), role: .parent)
    check(mixed.access == nil, "A mixed child/adult profile obtained contact account access")
    state.close(); forbidden.close()
}
